// sync_fifo: single-clock FIFO with wrap-bit pointers.
// DEPTH must be a power of two. FWFT=1 presents the head word on rd_data
// as soon as it is written (show-ahead); FWFT=0 returns it the cycle after rd_en.
module sync_fifo #(
  parameter int WIDTH     = 32,
  parameter int DEPTH     = 16,
  parameter int AF_THRESH = DEPTH - 2,  // almost_full when count >= this
  parameter bit FWFT      = 1'b0
) (
  input  logic             clk,
  input  logic             rst_n,
  input  logic             wr_en,
  input  logic [WIDTH-1:0] wr_data,
  input  logic             rd_en,
  output logic [WIDTH-1:0] rd_data,
  output logic             full,
  output logic             empty,
  output logic             almost_full,
  output logic [$clog2(DEPTH):0] count
);
  localparam int AW = $clog2(DEPTH);

  initial assert (DEPTH == (1 << AW)) else $fatal(1, "DEPTH must be a power of two");

  // Storage and pointers: one extra MSB (the wrap bit) tells full from empty.
  logic [WIDTH-1:0] mem [DEPTH];
  logic [AW:0]      wr_ptr, rd_ptr;

  wire do_write = wr_en && !full;
  wire do_read  = rd_en && !empty;

  // Write side
  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) wr_ptr <= '0;
    else if (do_write) begin
      mem[wr_ptr[AW-1:0]] <= wr_data;
      wr_ptr <= wr_ptr + 1'b1;
    end
  end

  // Read side
  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) rd_ptr <= '0;
    else if (do_read) rd_ptr <= rd_ptr + 1'b1;
  end

  // Status flags: same index, wrap bits differ => full; identical => empty.
  assign empty       = (wr_ptr == rd_ptr);
  assign full        = (wr_ptr[AW-1:0] == rd_ptr[AW-1:0]) &&
                       (wr_ptr[AW] != rd_ptr[AW]);
  assign count       = wr_ptr - rd_ptr;
  assign almost_full = (count >= AF_THRESH);

  // Read data path
  generate
    if (FWFT) begin : g_fwft
      // Show-ahead: the head word is visible combinationally.
      assign rd_data = mem[rd_ptr[AW-1:0]];
    end else begin : g_std
      // Standard: registered output, valid one cycle after rd_en.
      always_ff @(posedge clk) begin
        if (do_read) rd_data <= mem[rd_ptr[AW-1:0]];
      end
    end
  endgenerate

`ifndef SYNTHESIS
  // Checks: never write when full, never read when empty.
  always @(posedge clk) if (rst_n) begin
    assert (!(wr_en && full))  else $error("write while full dropped");
    assert (!(rd_en && empty)) else $error("read while empty ignored");
  end
`endif
endmodule
